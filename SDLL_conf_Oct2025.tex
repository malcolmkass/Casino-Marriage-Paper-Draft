\begin{frame} \frametitle{Effect of new casino development on Population and Family Dynamics}
	\begin{itemize}
		\item Local officials using new casinos to enrich local economies via economic inflows.
		\begin{itemize}
        \item Evidence of some short term benefits of employment and tax revenues. \parencite{humphreys2013new,wiley2011casino}
        \item Significant evidence of long term negative spillovers on the social strata. \parencite{grinols2006casinos}
        \end{itemize}
        \item []
        \item In addition, communities may wish to promote population growth. 
        \item Effects of casinos on population and family formation are not well understood.
        \item Only event analysis studies \parencite{moellman2013indian}
     \end{itemize}
    \end{frame}

\begin{frame}\frametitle{Related Work: Economic Activity and Family Formation}
    \begin{itemize}
      \item Family and community structures build social capital and improve outcomes for children from low-income families \parencite{chetty2026opportunity}
      \item Fertility as consumption: Positive income shocks increase fertility \parencite{kearney2023two, gallego2023baby}
      \item Fertility as investment: Positive income shocks increase savings; parents may view children as a way to smooth future consumption \parencite{goldin2021career,doepke2023economics}
    \end{itemize}
    \end{frame}

	\begin{itemize}
		\item Increased Crime: Casinos caused increases in aggravated assault, rape, robbery, larceny, burglary, auto theft \parencite{grinols2006casinos}
        \begin{itemize}
        \item Magnitude: 8.6\% of property crime and 12.6\% of violent crime in casino counties attributable to casinos 
        \end{itemize}
        \item Increases in Economic Outcomes
        \begin{itemize}
            \item Retail Property values \parencite{wiley2011casino}

        \end{itemize}
        \item Unique for this project: no preexisting casino presence
    \end{itemize}
\end{frame}

	\begin{itemize}
		\item Novel work on how a new casino entrant affects marriage, fertility, and population composition, setting akin to Sands' entry into Irving
        \item Unclear direction: Crime increases (-) vs. Employment gains (+)
        \item []
        \item [] Initial Results
        \item Casino $\rightarrow$ County level children population drops
        \item Casino $\rightarrow$ 2.01 percentage point decline in fertility rates for women, 3.17 drop for men
        \item Casino $\rightarrow$ drop in School enrollment and school age population, around 6 percentage point drop 3 years after opening
    \end{itemize}
\end{frame}


\section{Data and Methods}
\begin{frame} \frametitle{Explanatory variable of interest: 2003-2023 US county-level casino openings}
	\begin{itemize}
    \item Created a database of new casino openings by county and adjacent counties from 2003 to 2023 (to match timeframe of ACS data)
    \item Previous research finds increased economic activity for 1-2 yearsbefore the casino opening. 
    \begin{itemize}
    \item Time of treatment effect is two years before casino opening. 
    \item Event Study Analysis: Base outcome is the 3rd lag before opening.
    \end{itemize} 
\item []
        \item [] \textbf{Treatment:} New casino counties. Only includes new casino counties in the data time frame, no previous casinos or additional casinos. (19 total) 
        \item [] \textbf{Control:} No Casino or casino adjacent counties or previous existing nearby casino to prevent spillovers clouding effects (Akin to a Donut Diff in Diff design.)
\end{itemize}

\end{frame}

\begin{frame} \frametitle{Dependent Variables: population and family formation measures}
	\begin{itemize}
    \item ACS micro data from 2000-2023 
	\item Adult and Child Population data
    \begin{itemize}
        \item Total, Child Rearing (18 to 40), Children ($\le$ 17), School age (5-17)
        \item Residence duration: Broken down by gender
    \end{itemize}
    \item Family and fertility data: 
    \begin{itemize}
        \item marital status (base: not married currently)
        \item has children
        \item number of children
    \end{itemize}
    \item School age populations and enrollments
    \item To tease out treatment effects, we use two methods: Event Study and Staggered Difference and Differences (DiD).
	\end{itemize}
\end{frame}


\begin{frame} \frametitle{Treatment vs. Control county level differences}
	\begin{itemize}
    \item Notable aggregate differences between treatment casino counties vs. control counties
        \begin{itemize}
            \item 32.0\% vs 38.0\% married, 36.2\% vs. 41.0\% has children, 71 vs. 82 children per 100 ppl. 
            \item 63.1\% vs 70.8\% White 
            \item Treated counties skew more rural than control counties. 
        \end{itemize}
        \item Gender and Age are similar
    \end{itemize}
\end{frame}

\begin{frame}\frametitle{Method 1: Event Study}
    \begin{itemize}
    \item Compares outcomes \emph{before vs. after} an event (e.g., casino opening) using never-treated places as a benchmark.
    \item Lines up each county by \textbf{event time} (years relative to opening), then estimate the average difference at each event year.
    \item What you look for:
    \begin{itemize}
    \item No clear pre-trend / ``parallel trends''.
    \item Coefficients show the size and timing of the effect (immediate, gradual, temporary, persistent).
    \end{itemize}
    \end{itemize}
    \end{frame}

\begin{frame} \frametitle{Method 2: Staggered Difference and Differences (DiD)}
		\begin{itemize}
	\item Staggered Difference and Differences estimated with two-way fixed effects (FE) specification:
    \item [] \begin{equation}
Y_{ct} = \alpha + \beta \cdot \text{Post}_{ct} + \mathbf{X}_{ct}'\gamma +    \mu_c + \lambda_t + \varepsilon_{ct}
\end{equation}
    \begin{itemize}
        \item $Y_{ct}$ is the DV in county $c$ and year $t$ 
        \item $\text{Post}_{ct}$ is an indicator equal to one if county $c$ has an operating casino in year $t$
        \item $\mathbf{X}_{ct}$ includes demographic controls 
        \item $\mu_c$ and $\lambda_t$ are county and year fixed effects
        \item Standard errors clustered at the county level
        \item Weighted by overall population (low population county averages do not hold the same weight as densely populated counties) 
	\end{itemize}
\end{itemize}
\end{frame}

\begin{frame} \frametitle{Staggered Difference and Differences (DiD):}
	\begin{itemize}
    \item Causal technique to estimate treatment effects (new casino)
     \item In many applications, we want to estimate the effect of a policy or economic change
    \item However, the policy assignment is not necessarily uncorrelated with group characteristics
    \begin{itemize}
        \item A new casino may be located specifically in treated areas with declining population and family formation
        \item DiD uses, as a control, areas that were never treated 
    \end{itemize}
    \item The treatment effect (effect of the casino) is the difference in the DV of the treated areas minus the difference in the DV of the control areas.
    \item Fixed Effects estimation accounts for county and time-specific effects that may confound results 
\end{itemize}
\end{frame}

	\begin{itemize}

    \item Need to employ staggered Diff in Diff corrections, specifically Calloway and Sant'Anna method.
